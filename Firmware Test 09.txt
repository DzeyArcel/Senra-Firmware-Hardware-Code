void log(String msg) {
  Serial.println("[LOG] " + msg);
}

String isoTime() {
  time_t now;
  time(&now);
  struct tm* t = gmtime(&now);

  char buf[30];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", t);
  return String(buf);
}

int batteryLevel() {
  return random(75, 95);
}

// ============================================================
// EEPROM
// ============================================================
void saveWiFi(String ssid, String pass) {
  for (int i = 0; i < 32; i++) EEPROM.write(SSID_ADDR + i, i < ssid.length() ? ssid[i] : 0);
  for (int i = 0; i < 32; i++) EEPROM.write(PASS_ADDR + i, i < pass.length() ? pass[i] : 0);
  EEPROM.commit();
  log("EEPROM saved WiFi credentials");
}

void clearWiFi() {
  for (int i = 0; i < 64; i++) EEPROM.write(SSID_ADDR + i, 0);
  EEPROM.commit();
  log("EEPROM cleared — WiFi wiped");
}

String readSSID() {
  char b[32];
  for (int i = 0; i < 32; i++) b[i] = EEPROM.read(SSID_ADDR + i);
  return String(b);
}

String readPASS() {
  char b[32];
  for (int i = 0; i < 32; i++) b[i] = EEPROM.read(PASS_ADDR + i);
  return String(b);
}

// ============================================================
// AP MODE PAGES
// ============================================================
void handleRoot() {
  log("AP Page visited");
  server.send(200, "text/html",
    "<h1>SENRA WiFi Setup</h1>"
    "<form action='/setwifi'>"
    "SSID:<input name='s'><br>"
    "PASS:<input name='p'><br>"
    "<input type='submit' value='Save'></form>");
}

void handleSetWiFi() {
  String ssid = server.arg("s");
  String pass = server.arg("p");

  log("WiFi received: " + ssid);

  if (ssid.length() > 0) saveWiFi(ssid, pass);

  server.send(200, "text/html", "<h1>Saved! Restarting...</h1>");
  delay(1500);
  ESP.restart();
}

void startAP() {
  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);

  server.on("/", handleRoot);
  server.on("/setwifi", handleSetWiFi);
  server.begin();

  log("AP mode enabled (OPEN) SSID=SENRA_SETUP");
}

// ============================================================
// WIFI CONNECT
// ============================================================
bool connectWiFi() {
  String ssid = readSSID();
  String pass = readPASS();

  if (ssid.length() < 1) {
    log("No WiFi saved → AP Mode");
    startAP();
    return false;
  }

  log("Connecting WiFi: " + ssid);

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid.c_str(), pass.c_str());

  unsigned long start = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - start < 9000) {
    delay(400);
    Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    log("WiFi Connected!");
    return true;
  }

  log("WiFi failed → AP Mode");
  startAP();
  return false;
}

// ============================================================
// GPS
// ============================================================
void readGPS() {
  while (GPS_Serial.available()) gps.encode(GPS_Serial.read());

  if (gps.location.isValid()) {
    lastLat = gps.location.lat();
    lastLng = gps.location.lng();
  }
}

// ============================================================
// SIM800
// ============================================================
bool simAvailable() {
  SIM800.println("AT+CSQ");
  delay(200);

  if (!SIM800.available()) return false;

  String r = SIM800.readString();
  return r.indexOf("99") == -1;
}

void sendSMS(String number, String msg) {
  log("Sending SMS to " + number);

  SIM800.println("AT+CMGF=1");
  delay(200);

  SIM800.print("AT+CMGS=\"");
  SIM800.print(number);
  SIM800.println("\"");

  delay(200);

  SIM800.print(msg);
  SIM800.write(26);

  delay(500);
}

// ============================================================
// FIRESTORE — CHECK ADMIN COMMAND(S)
// ============================================================
void checkAdminCommands() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  http.begin(deviceURL);
  int code = http.GET();

  if (code != 200) {
    log("AdminCommand READ failed");
    http.end();
    return;
  }

  String payload = http.getString();
  http.end();

  if (payload.indexOf("wifi_reset") != -1) {
    log("🔥 WiFi RESET REQUESTED FROM APP!");
    wifiResetRequested = true;
  }
}

// ============================================================
// FIRESTORE — CLEAR ADMIN COMMAND
// ============================================================
void clearAdminCommand() {
  HTTPClient http;

  String patch =
    "{ \"fields\": {\"adminCommand\": {\"stringValue\": \"\"}}}";

  String url = deviceURL +
    "&updateMask.fieldPaths=adminCommand";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  http.PATCH(patch);
  http.end();
}

// ============================================================
// FIRESTORE — UPDATE DEVICE DOC
// ============================================================
void updateDeviceDoc(bool fallDetected, String fallStatus) {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;

  String url = deviceURL +
    "&updateMask.fieldPaths=status"
    "&updateMask.fieldPaths=batteryLevel"
    "&updateMask.fieldPaths=fallDetected"
    "&updateMask.fieldPaths=fallStatus"
    "&updateMask.fieldPaths=currentAlertId"
    "&updateMask.fieldPaths=lat"
    "&updateMask.fieldPaths=lng"
    "&updateMask.fieldPaths=lastSync";

  String json = "{ \"fields\": {";

  json += "\"status\":{\"stringValue\":\"online\"},";
  json += "\"batteryLevel\":{\"integerValue\":\"" + String(batteryLevel()) + "\"},";
  json += "\"fallDetected\":{\"booleanValue\":" + String(fallDetected ? "true" : "false") + "},";
  json += "\"fallStatus\":{\"stringValue\":\"" + fallStatus + "\"},";
  json += "\"currentAlertId\":{\"stringValue\":\"" + currentAlertId + "\"},";

  if (lastLat != 0) json += "\"lat\":{\"doubleValue\":" + String(lastLat,6) + "},";
  if (lastLng != 0) json += "\"lng\":{\"doubleValue\":" + String(lastLng,6) + "},";

  json += "\"lastSync\":{\"timestampValue\":\"" + isoTime() + "\"}";
  json += "} }";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  http.PATCH(json);
  http.end();

  log("Device updated → fallStatus=" + fallStatus);
}

// ============================================================
// FIRESTORE — CREATE ALERT
// ============================================================
void createAlert() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;

  String url =
    "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID +
    "/databases/(default)/documents/alerts?key=" + API_KEY;

  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  String mapURL =
    "https://www.google.com/maps?q=" + String(lastLat,6) + "," + String(lastLng,6);

  String json = "{ \"fields\": {";
  json += "\"deviceId\": {\"stringValue\": \"" + DEVICE_ID + "\"},";
  json += "\"lat\":       {\"doubleValue\": " + String(lastLat,6) + "},";
  json += "\"lng\":       {\"doubleValue\": " + String(lastLng,6) + "},";
  json += "\"location\":  {\"stringValue\": \"" + mapURL + "\"},";
  json += "\"mapURL\":    {\"stringValue\": \"" + mapURL + "\"},";
  json += "\"fallType\":  {\"stringValue\": \"Fall Detected\"},";
  json += "\"delivered\": {\"booleanValue\": false},";
  json += "\"status\":    {\"stringValue\": \"pending\"},";
  json += "\"timestamp\": {\"timestampValue\":\"" + isoTime() + "\"}";
  json += "} }";

  int code = http.POST(json);
  if (code == 200) {
    String payload = http.getString();
    int pos = payload.indexOf("/alerts/");
    if (pos != -1) {
      int start = pos + 8;
      int end = payload.indexOf("\"", start);
      currentAlertId = payload.substring(start, end);
      log("✔ ALERT CREATED: " + currentAlertId);
    }
  }

  http.end();
}

// ============================================================
// CANCEL ALERT
// ============================================================
void cancelAlert() {
  if (currentAlertId == "" || WiFi.status() != WL_CONNECTED) return;

  log("Cancelling Alert…");

  HTTPClient http;

  String url =
    "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID +
    "/databases/(default)/documents/alerts/" + currentAlertId +
    "?key=" + API_KEY;

  String json =
    "{ \"fields\": {"
    "\"status\": {\"stringValue\": \"cancelled_by_device\"},"
    "\"cancelled_at\": {\"timestampValue\": \"" + isoTime() + "\"}"
    "} }";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  http.PATCH(json);
  http.end();

  currentAlertId = "";
  updateDeviceDoc(false, "cancelled_by_device");
}

// ============================================================
// FALL DETECTION — V14 ULTRA-SENSITIVE
// ============================================================
void checkFall() {
  sensors_event_t a, g, t;
  mpu.getEvent(&a, &g, &t);

  // Normalize to G-force
  float ax = a.acceleration.x / 9.81;
  float ay = a.acceleration.y / 9.81;
  float az = a.acceleration.z / 9.81;
  float totalG = sqrt(ax*ax + ay*ay + az*az);

  // Gyro (deg/sec)
  float rotX = abs(g.gyro.x);
  float rotY = abs(g.gyro.y);
  float rotZ = abs(g.gyro.z);
  float rotMax = max(rotX, max(rotY, rotZ));  

  // ----------- ULTRA SENSITIVE THRESHOLDS ----------
  const float IMPACT_G = 1.9;      // VERY sensitive
  const float ROT_SPIKE = 80.0;    // detects sideways & twisting falls
  const float STILL_G = 1.55;      // slight margin for noise

  // ----------- TRIGGER FALL (impact OR rotation) ----------
  if (!fallTriggered && (totalG > IMPACT_G || rotMax > ROT_SPIKE)) {
    fallTriggered = true;
    fallTime = millis();
    log("⚠ Impact/Rotation detected! G=" + String(totalG) +
        " | rot=" + String(rotMax));
  }

  // ----------- VERIFY AFTER 200ms ----------
  if (fallTriggered && millis() - fallTime > 200) {

    if (totalG < STILL_G) {
      log("✔ CONFIRMED FALL — Sending alert");

      updateDeviceDoc(true, "pending");

      // CANCEL WINDOW (8 seconds)
      for (int i = 0; i < 8; i++) {
        if (digitalRead(BUTTON_PIN) == LOW) {
          log("Cancel button pressed → CANCELLED");
          cancelAlert();
          fallTriggered = false;
          return;
        }

        // Buzzer signal
        digitalWrite(BUZZER_PIN, HIGH); delay(180);
        digitalWrite(BUZZER_PIN, LOW);  delay(180);
      }

      // SEND ALERT
      createAlert();
      updateDeviceDoc(true, "pending");

      // SMS fallback
      if (simAvailable()) {
        String msg =
          "⚠ SENRA ALERT\nDevice: " + DEVICE_ID +
          "\nLat:" + String(lastLat,6) +
          "\nLng:" + String(lastLng,6);

        sendSMS("+639123456789", msg);
      }
    }

    fallTriggered = false;
  }
}


// ============================================================
// SETUP
// ============================================================
void setup() {
  Serial.begin(115200);
  EEPROM.begin(EEPROM_SIZE);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  Wire.begin(21, 22);

  GPS_Serial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  SIM800.begin(9600, SERIAL_8N1, SIM_RX, SIM_TX);

  if (!mpu.begin()) {
    log("❌ MPU6050 NOT FOUND");
  }

  configTime(0, 0, "pool.ntp.org");

  connectWiFi();
}

// ============================================================
// LOOP
// ============================================================
void loop() {
  server.handleClient();

  if (WiFi.status() == WL_CONNECTED) {
    checkAdminCommands();

    if (wifiResetRequested) {
      log("🔥 EXECUTING WIFI RESET…");
      updateDeviceDoc(false, "wifi_resetting");
      clearWiFi();
      delay(500);
      ESP.restart();
    }
  }

  readGPS();
  checkFall();

  // Heartbeat
  if (millis() - lastSyncSend > 20000 && WiFi.status() == WL_CONNECTED) {
    updateDeviceDoc(false, "none");
    lastSyncSend = millis();
  }
}
