// ================= TIME =================
const char* isoTime() {
  static char buf[32];
  time_t now;
  time(&now);
  struct tm* t = gmtime(&now);
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", t);
  return buf;
}

// ================= HTTP SAFETY =================
bool canPerformHttpRequest() {
  if (fallStateLocked) {
    Serial.println("⛔ HTTP blocked: fall state locked");
    return false;
  }
  
  if (httpRequestInProgress && millis() - httpStartTime > HTTP_WATCHDOG_TIMEOUT) {
    Serial.println("⚠️ HTTP watchdog reset (stuck request)");
    httpRequestInProgress = false;
  }

  if (httpRequestInProgress) {
    Serial.println("⛔ HTTP blocked: request in progress");
    return false;
  }
  
  if (millis() - lastHttpEndTime < HTTP_COOLDOWN_MS) {
    delay(HTTP_COOLDOWN_MS - (millis() - lastHttpEndTime));
  }
  
  return true;
}

void markHttpRequestStart() {
  httpRequestInProgress = true;
  httpStartTime = millis();
}

void markHttpRequestEnd() {
  httpRequestInProgress = false;
  lastHttpEndTime = millis();
}

void safeHttpEnd(HTTPClient& http) {
  http.end();
  markHttpRequestEnd();
  delay(HTTP_COOLDOWN_MS);
}

void saveField(int addr, String v) {
  for (int i = 0; i < 32; i++)
    EEPROM.write(addr + i, i < v.length() ? v[i] : 0);

}

String readField(int addr) {
  char b[33];
  for (int i = 0; i < 32; i++) b[i] = EEPROM.read(addr + i);
  b[32] = '\0';

}

// UPGRADE 1: Improved battery monitoring with ADC configuration
// Changed samples from 5 to 10, added ADC attenuation and resolution in setup()
int readBatteryPercent() {
  const int samples = 10;  // UPGRADE: Increased from 5 to 10 for better stability
  float readings[samples];
  
  for (int i = 0; i < samples; i++) {
    // UPGRADE: Changed calculation order for better precision
    readings[i] = analogRead(BATTERY_PIN) * 3.3 / 4095.0 * 2.0;
    delayMicroseconds(800);
  }
  
  // Median filter (unchanged logic)
  for (int i = 0; i < samples-1; i++) {
    for (int j = 0; j < samples-i-1; j++) {
      if (readings[j] > readings[j+1]) {
        float temp = readings[j];
        readings[j] = readings[j+1];
        readings[j+1] = temp;
      }
    }
  }
  
  static float smooth = 0;
float v = readings[samples/2];

smooth = smooth * 0.7 + v * 0.3;
v = smooth;
  return constrain(map(v * 100, 330, 420, 0, 100), 0, 100);
}

String extractStringField(const String& json, const String& field) {
  int p = json.indexOf("\"" + field + "\"");
  if (p == -1) return "";

  int v = json.indexOf("stringValue", p);
  if (v == -1) return "";

  int q1 = json.indexOf("\"", v + 12);
  int q2 = json.indexOf("\"", q1 + 1);
  if (q1 == -1 || q2 == -1) return "";

  return json.substring(q1 + 1, q2);
}

// FIXED: All variables declared at top before any goto
bool sendSMS_Safe(const String& number, const String& message) {
  // All variables declared at top - before any goto
  bool result = false;
  bool promptOK = false;
  unsigned long t;
  const size_t chunkSize = 128;
  size_t msgLen = 0;
  size_t sent = 0;
  size_t remaining = 0;
  size_t toSend = 0;
  enum { WAIT_CMGS, WAIT_OK, GOT_ERROR } state = WAIT_CMGS;
  String line;
  
  if (number.length() < 10) {
    Serial.println("❌ SMS aborted: invalid number");
    return false;
  }

  if (smsInProgress) {
    Serial.println("⏳ SMS already in progress");
    return false;
  }

  smsInProgress = true;

  Serial.print("📲 Sending SMS to ");
  Serial.println(number);

  while (SIM800.available()) SIM800.read();
  
  SIM800.print("AT+CMGS=\"");
  SIM800.print(number);
  SIM800.println("\"");

  t = millis();
  
  while (millis() - t < 5000) {
    if (SIM800.available()) {
      char c = SIM800.read();
      if (c == '>') {
        promptOK = true;
        break;
      }
    }
    delay(10);
  }

  if (!promptOK) {
    Serial.println("❌ SMS failed: no prompt");
    goto sms_cleanup;
  }

  msgLen = message.length();
  sent = 0;
  
  while (sent < msgLen) {
    remaining = msgLen - sent;
    toSend = (remaining > chunkSize) ? chunkSize : remaining;
    SIM800.print(message.substring(sent, sent + toSend));
    sent += toSend;
    delay(50);
  }
  
  SIM800.write(26);

  t = millis();
  state = WAIT_CMGS;
  
  while (millis() - t < 15000) {
    if (SIM800.available()) {
      line = SIM800.readStringUntil('\n');
      line.trim();
      
      if (line.indexOf("+CMGS:") != -1) {
        state = WAIT_OK;
      } else if (line.indexOf("OK") != -1 && state == WAIT_OK) {
        result = true;
        break;
      } else if (line.indexOf("ERROR") != -1 || line.indexOf("+CMS ERROR") != -1) {
        state = GOT_ERROR;
        Serial.println("❌ SMS ERROR: " + line);
        break;
      }
    }
    delay(10);
  }

sms_cleanup:
  smsInProgress = false;
  
  if (result) {
    Serial.println("✅ SMS SENT SUCCESSFULLY");
  } else {
    Serial.println("❌ SMS SEND FAILED");
  }
  
  while (SIM800.available()) SIM800.read();
  
  return result;
}

void processSMSQueue() {
  if (smsInProgress) return;
  
  for (int i = 0; i < 2; i++) {
    if (smsRetryQueue[i].pending) {
      if (millis() - smsRetryQueue[i].lastAttempt < SMS_RETRY_DELAY) {
        continue;
      }
      
      if (smsRetryQueue[i].retries >= MAX_SMS_RETRIES) {
        Serial.printf("❌ SMS %d: Max retries exceeded\n", i);
        smsRetryQueue[i].pending = false;
        continue;
      }
      
      smsRetryQueue[i].lastAttempt = millis();
      smsRetryQueue[i].retries++;
      
      Serial.printf("📲 SMS %d: Attempt %d/%d\n", i, smsRetryQueue[i].retries, MAX_SMS_RETRIES);
      
      if (sendSMS_Safe(smsRetryQueue[i].number, smsRetryQueue[i].message)) {
        smsRetryQueue[i].pending = false;
        Serial.printf("✅ SMS %d: Delivered\n", i);
      } else {
        Serial.printf("⚠️ SMS %d: Failed, will retry\n", i);
      }
      
      return;
    }
  }
}

// FIXED: All json.indexOf changed to body.indexOf, all http.begin use full URL
void pollCloud() {
  if (!tcpipReady) return;
  if (waitingCloudClear) {
    Serial.println("⛔ Waiting for cloud to clear reset flags");
    return;
  }
  if (wifiResetHandledThisBoot) return;
  if (fallState == ALERT_PENDING || fallState == POST_ALERT) {
    Serial.println("⛔ Cloud commands blocked (active fall)");
    return;
  }
  if (WiFi.status() != WL_CONNECTED) return;
  if (wifiConnectedAt == 0) return;
  if (millis() - wifiConnectedAt < WIFI_SETTLE_TIME) return;
  if (getWifiResetFlag()) return;
  
  if (!canPerformHttpRequest()) {
    Serial.println("⛔ pollCloud blocked: HTTP busy");
    return;
  }

  unsigned long interval = POLL_INTERVAL_MS;
  if (consecutiveHttpErrors > 2) {
    interval += (consecutiveHttpErrors * ERROR_RETRY_BACKOFF_MS);
    Serial.print("⏳ Error backoff active, interval: ");
    Serial.println(interval);
  }
  
  if (millis() - lastCloudCheck < interval) return;

  markHttpRequestStart();
  
  WiFiClientSecure client;
  setupSecureClient(client);
  
  HTTPClient http;
  http.useHTTP10(true);
  
  // Always use full URL string - no IP begin() overload
  http.begin(client, deviceURL);
  
  http.setTimeout(4000);
  http.setConnectTimeout(5000);
  
  int code = http.GET();
  lastCloudCheck = millis();

  if (code != HTTP_CODE_OK) {
    Serial.print("☁️ pollCloud HTTP error: ");
    Serial.println(code);
    consecutiveHttpErrors++;
    
    safeHttpEnd(http);
    return;
  }

  // UPGRADE 6: Reserve memory before storing HTTP response to reduce heap fragmentation
  String body;
  body.reserve(512);
  body = http.getString();
  
  safeHttpEnd(http);
  consecutiveHttpErrors = 0;
  lastPollSuccessTime = millis();

  // Cache IP for future use (but still use URL string)
  if (!firestoreIPCached && WiFi.hostByName("firestore.googleapis.com", firestoreIP)) {
    firestoreIPCached = true;
    Serial.print("🔒 Firestore IP cached: ");
    Serial.println(firestoreIP);
  }

  if (body.length() < 50) {
    Serial.println("⚠️ pollCloud: response too small");
    return;
  }

  // FIXED: Use body.indexOf() not json.indexOf()
  int op = body.indexOf("\"ownerPhone\"");
  if (op != -1) {
    int v = body.indexOf("stringValue", op);
    int q1 = body.indexOf("\"", v + 12);  // FIXED: was json.indexOf
    int q2 = body.indexOf("\"", q1 + 1);  // FIXED: was json.indexOf

    if (q1 != -1 && q2 != -1) {
      String newOwner = body.substring(q1 + 1, q2);
      newOwner.trim();

      if (newOwner.length() >= 10 && newOwner != ownerPhone) {
        ownerPhone = newOwner;
        saveField(PAIRED_PHONE_ADDR, ownerPhone);
        Serial.print("📞 Owner phone updated: ");
        Serial.println(ownerPhone);
      }
    }
  }

  int ep = body.indexOf("\"emergencyPhone\"");
  if (ep != -1) {
    int v = body.indexOf("stringValue", ep);
    int q1 = body.indexOf("\"", v + 12);  // FIXED: was json.indexOf
    int q2 = body.indexOf("\"", q1 + 1);  // FIXED: was json.indexOf

    if (q1 != -1 && q2 != -1) {
      String newEmergency = body.substring(q1 + 1, q2);
      newEmergency.trim();

      bool valid = newEmergency.length() >= 10;
      for (char c : newEmergency) {
        if (!(isDigit(c) || c == '+')) {
          valid = false;
          break;
        }
      }

      if (valid && newEmergency != emergencyPhone) {
        emergencyPhone = newEmergency;
        saveField(EMERGENCY_PHONE_ADDR, emergencyPhone);
        Serial.print("🚑 Emergency phone updated: ");
        Serial.println(emergencyPhone);
      }
    }
  }

  if (body.indexOf("\"paired_to\"") != -1) {
    int p = body.indexOf("\"paired_to\"");
    int v = body.indexOf("stringValue", p);

    if (v != -1) {
      int q1 = body.indexOf("\"", v + 12);
      int q2 = body.indexOf("\"", q1 + 1);

      if (q1 != -1 && q2 != -1 && q2 > q1) {
        String pairedTo = body.substring(q1 + 1, q2);
        pairedTo.trim();
        handleUnlink(pairedTo);
      }
    }
  }

  bool wifiResetRequested = false;
  bool wifiResetAcked = false;

  int r = body.indexOf("\"wifiReset\"");
  if (r != -1) {
    int rb = body.indexOf("booleanValue", r);
    if (rb != -1 && body.indexOf("true", rb) != -1) {
      wifiResetRequested = true;
    }
  }

  int a = body.indexOf("\"wifiResetAck\"");
  if (a != -1) {
    int ab = body.indexOf("booleanValue", a);
    if (ab != -1 && body.indexOf("true", ab) != -1) {
      wifiResetAcked = true;
    }
  }

  if (!wifiResetRequested || wifiResetAcked) return;

  Serial.println("🔥 CLOUD WIFI RESET CONFIRMED");

  markDeviceResetting();
  delay(800);

  wifiResetHandledThisBoot = true;
  waitingCloudClear = true;

  if (canPerformHttpRequest()) {
    markHttpRequestStart();
    WiFiClientSecure ackClient;
    setupSecureClient(ackClient);
    HTTPClient ack;
    ack.useHTTP10(true);
    
    // FIXED: Use full URL, not IP overload
    String ackUrl = deviceURL + "&updateMask.fieldPaths=wifiResetAck";
    ack.begin(ackClient, ackUrl);
    
    ack.addHeader("Content-Type", "application/json");
    ack.PATCH("{\"fields\":{\"wifiResetAck\":{\"booleanValue\":true}}}");
    safeHttpEnd(ack);
    
    delay(100);
    
    markHttpRequestStart();
    WiFiClientSecure clearClient;
    setupSecureClient(clearClient);
    HTTPClient clear;
    clear.useHTTP10(true);
    
    // FIXED: Use full URL, not IP overload
    String clearUrl = deviceURL + "&updateMask.fieldPaths=wifiReset";
    clear.begin(clearClient, clearUrl);
    
    clear.addHeader("Content-Type", "application/json");
    clear.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
    safeHttpEnd(clear);
  }

  delay(300);
  saveField(SSID_ADDR, "");
  saveField(PASS_ADDR, "");
  setWifiResetFlag(true);

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2000);

  Serial.println("🔁 REBOOTING TO AP MODE");
  ESP.restart();
}

// ================= FALL ENGINE =================
bool initMPU() {
  Wire.begin(21, 22);
  Wire.setClock(100000);
  
  if (!mpu.begin()) {
    Serial.println("❌ MPU6050 NOT FOUND");
    mpuInitialized = false;
    return false;
  }
  
  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
  
  mpuInitialized = true;
  Serial.println("🧠 MPU initialized");
  return true;
}

void checkFall() {
  static unsigned long postAlertAt = 0;
  static bool alertReported = false;
  static float filteredG = 1.0;

  if (!mpuInitialized && millis() - lastMpuCheck > 30000) {
    lastMpuCheck = millis();
    initMPU();
    return;
  }
  
  if (!mpuInitialized) return;
  
  sensors_event_t a, g, t;
  
  Wire.setTimeOut(500);
  if (!mpu.getEvent(&a, &g, &t)) {
    Serial.println("⚠️ MPU read failed - scheduling re-init");
    mpuInitialized = false;
    return;
  }

  if (!mpuReady) {
    if (mpuInitAt == 0) mpuInitAt = millis();

    if (millis() - mpuInitAt > 800) {
      mpuReady = true;
      Serial.println("🧠 MPU ready");
    } else {
      return;
    }
  }

  if (fallState == POST_ALERT) {
    if (postAlertAt == 0) {
      postAlertAt = millis();
    }

    if (millis() - postAlertAt > 15000) {
      postAlertAt = 0;
      fallStateLocked = true;
      fallState = IDLE;
      fallStateLocked = false;
      alertReported = false;
      smsSentThisFall = false;
      smsInProgress = false;
    }
    return;
  }

  float totalG = sqrt(
    sq(a.acceleration.x / 9.81) +
    sq(a.acceleration.y / 9.81) +
    sq(a.acceleration.z / 9.81)
  );

  filteredG = 0.85 * filteredG + 0.15 * totalG;

  unsigned long now = millis();

  if (fallState == IDLE) {
    if (filteredG > IMPACT_G || abs(g.gyro.x) > ROT_SPIKE) {
      fallStateLocked = true;
      fallState = CANDIDATE;
      fallTime = now;
      fallStateLocked = false;
    }
    return;
  }

  if (fallState == CANDIDATE) {
    if (now - fallTime > 1000) {
      fallStateLocked = true;
      fallState = IDLE;
      fallStateLocked = false;
      return;
    }

    if (now - fallTime > CONFIRM_DELAY) {
      fallStateLocked = true;
      if (filteredG < STILL_G) {
        fallState = ALERT_PENDING;
        alertStart = now;
        alertReported = false;
        updateDeviceDoc(true, "pending");
      } else {
        fallState = IDLE;
      }
      fallStateLocked = false;
    }
    return;
  }

  if (fallState == ALERT_PENDING) {
    if (now - lastBuzz > BUZZ_INTERVAL) {
      digitalWrite(BUZZER_PIN, !digitalRead(BUZZER_PIN));
      lastBuzz = now;
    }

    static unsigned long lastButtonCheck = 0;
    static int buttonStableCount = 0;
    
    if (millis() - lastButtonCheck > 50) {
      lastButtonCheck = millis();
      
      if (digitalRead(BUTTON_PIN) == LOW) {
        buttonStableCount++;
        if (buttonStableCount >= 3) {
          fallStateLocked = true;
          digitalWrite(BUZZER_PIN, LOW);
          updateDeviceDoc(false, "cancelled");
          fallState = IDLE;
          fallStateLocked = false;
          buttonStableCount = 0;
        }
      } else {
        buttonStableCount = 0;
      }
    }

    if (now - alertStart > CANCEL_WINDOW) {
      digitalWrite(BUZZER_PIN, LOW);

      if (!alertReported) {
        if (WiFi.status() == WL_CONNECTED) {
          createAlert();
        }
        
        // Queue SMS instead of immediate send
        if (!smsSentThisFall) {
          // UPGRADE 3: Protect GPS data reads from race conditions
          double alertLat = 0;
          double alertLng = 0;
          
          noInterrupts();
          alertLat = lastLat;
          alertLng = lastLng;
          interrupts();
          
          char msg[160];
          if (gpsDataFresh && (alertLat != 0 || alertLng != 0)) {
            snprintf(msg, sizeof(msg),
              "SENRA FALL ALERT\nDevice: %s\nLoc: https://maps.google.com/?q=%.6f,%.6f",
              DEVICE_ID.c_str(), alertLat, alertLng);
          } else {
            snprintf(msg, sizeof(msg),
              "SENRA FALL ALERT\nDevice: %s\nLocation unavailable",
              DEVICE_ID.c_str());
          }
          
          int queueIndex = 0;
          
          // UPGRADE 2: Prevent SMS queue overflow with bounds checking
          if (ownerPhone.length() > 10 && queueIndex < 2) {
            smsRetryQueue[queueIndex].pending = true;
            smsRetryQueue[queueIndex].number = ownerPhone;
            smsRetryQueue[queueIndex].message = String(msg);
            smsRetryQueue[queueIndex].retries = 0;
            smsRetryQueue[queueIndex].lastAttempt = 0;
            queueIndex++;
          }
          
          // UPGRADE 2: Prevent SMS queue overflow with bounds checking
          if (emergencyPhone.length() > 10 && emergencyPhone != ownerPhone && queueIndex < 2) {
            smsRetryQueue[queueIndex].pending = true;
            smsRetryQueue[queueIndex].number = emergencyPhone;
            smsRetryQueue[queueIndex].message = String(msg);
            smsRetryQueue[queueIndex].retries = 0;
            smsRetryQueue[queueIndex].lastAttempt = 0;
          }
          
          smsSentThisFall = true;
        }
        
        alertReported = true;
      }

      fallState = POST_ALERT;
    }
  }
}

// ================= WIFI =================
bool connectWiFi() {
  WiFi.mode(WIFI_STA);

  String ssid = readSSID();
  ssid.trim();
  String pass = readPASS();
  pass.trim();

  Serial.println("----- WIFI DEBUG -----");
  Serial.print("Stored SSID: [");
  Serial.print(ssid);
  Serial.println("]");
  Serial.print("Stored PASS length: ");
  Serial.println(pass.length());

  if (ssid.length() == 0) {
    Serial.println("No SSID found → Starting AP mode");
    startAP();
    return false;
  }

  // UPGRADE 5: Improve WiFi signal strength before connecting
  WiFi.setTxPower(WIFI_POWER_19_5dBm);
  WiFi.setSleep(false);
  
  Serial.println("Connecting to WiFi...");
  WiFi.begin(ssid.c_str(), pass.c_str());

  unsigned long t = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t < 20000) {
    delay(500);
    Serial.print(".");
  }
  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("✅ WIFI CONNECTED");
    Serial.print("IP: ");
    Serial.println(WiFi.localIP());

    wifiConnectedAt = millis();
    server.on("/setcaregiver", handleSetCaregiver);
    server.begin();
    return true;
  }

  Serial.println("❌ WIFI FAILED");
  Serial.print("WiFi status code: ");
  Serial.println(WiFi.status());

  startAP();
  return false;
}

// ================= SETUP =================
void setup() {
  tcpipReady = false;
  wifiConnectedAt = 0;
  waitingCloudClear = false;
  wifiResetHandledThisBoot = false;
  phoneSyncedThisBoot = false;

  Serial.begin(115200);
  delay(1200);
  EEPROM.begin(EEPROM_SIZE);

  Serial.println("\n=== SENRA HARD BOOT ===");

  ESP_ERROR_CHECK(esp_task_wdt_init(30, true));
  ESP_ERROR_CHECK(esp_task_wdt_add(NULL));
  Serial.println("✅ Watchdog enabled (30s)");

  ownerPhone = readField(PAIRED_PHONE_ADDR);
  ownerPhone.trim();

  bool ownerValid = ownerPhone.length() >= 10;
  for (char c : ownerPhone) {
    if (!(isDigit(c) || c == '+')) {
      ownerValid = false;
      break;
    }
  }

  if (!ownerValid) {
    Serial.println("⚠️ Invalid owner phone — clearing");
    ownerPhone = "";
    saveField(PAIRED_PHONE_ADDR, "");
  } else {
    Serial.print("📞 Loaded paired phone: ");
    Serial.println(ownerPhone);
  }

  emergencyPhone = readField(EMERGENCY_PHONE_ADDR);
  emergencyPhone.trim();

  bool emergencyValid = emergencyPhone.length() >= 10;
  for (char c : emergencyPhone) {
    if (!(isDigit(c) || c == '+')) {
      emergencyValid = false;
      break;
    }
  }

  if (!emergencyValid) {
    Serial.println("⚠️ Invalid emergency phone — clearing");
    emergencyPhone = "";
    saveField(EMERGENCY_PHONE_ADDR, "");
  } else {
    Serial.print("🚑 Loaded emergency phone: ");
    Serial.println(emergencyPhone);
  }

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2500);

  if (getWifiResetFlag()) {
    Serial.println("🔥 WIFI RESET FLAG FOUND — FORCING AP");

    WiFi.persistent(false);
    WiFi.setAutoReconnect(false);
    WiFi.setSleep(false);

    WiFi.mode(WIFI_AP);
    WiFi.softAP("SENRA_SETUP");

    server.on("/", handleRoot);
    server.on("/setwifi", handleSetWiFi);
    server.begin();

    while (true) {
      server.handleClient();
      delay(10);
      esp_task_wdt_reset();
    }
  }

  Serial.println("Normal boot...");

  WiFi.persistent(false);
  WiFi.setAutoReconnect(true);
  
  // UPGRADE 4: Disable WiFi power-saving mode for TLS stability
  WiFi.setSleep(false);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(BATTERY_PIN, INPUT);
  
  // UPGRADE 1: Configure ADC for better battery monitoring stability
  analogSetPinAttenuation(BATTERY_PIN, ADC_11db);
  analogReadResolution(12);
  
  initMPU();
  mpuInitAt = millis();
  mpuReady = false;

  GPS_Serial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  SIM800.begin(9600, SERIAL_8N1, SIM_RX, SIM_TX);
  initSIM();

  configTime(0, 0, "pool.ntp.org", "time.google.com");

  connectWiFi();
  
  if (WiFi.status() == WL_CONNECTED) {
    if (WiFi.hostByName("firestore.googleapis.com", firestoreIP)) {
      firestoreIPCached = true;
      Serial.print("🔒 Firestore IP cached: ");
      Serial.println(firestoreIP);
    }
  }
}

// ================= LOOP =================
void loop() {
  esp_task_wdt_reset();
  
  server.handleClient();
  esp_task_wdt_reset();
  
  readGPS();
  esp_task_wdt_reset();
  
  checkFall();
  esp_task_wdt_reset();
  
  processSMSQueue();
  esp_task_wdt_reset();

  static bool wasConnected = false;

  if (WiFi.status() != WL_CONNECTED) {
    if (wasConnected) {
      Serial.println("🔴 WiFi lost → marking OFFLINE");
      markDeviceOffline();
      wasConnected = false;
      tcpipReady = false;
      firestoreIPCached = false;
    }
    
    if (millis() - lastWiFiReconnectAttempt > WIFI_RECONNECT_INTERVAL * (1 + wifiReconnectAttempts)) {
      Serial.println("🔄 Attempting WiFi reconnect...");
      WiFi.reconnect();
      lastWiFiReconnectAttempt = millis();
      wifiReconnectAttempts = min(wifiReconnectAttempts + 1, 5);
    }
    
    delay(100);
    return;
  }

  if (!wasConnected) {
    Serial.println("🟢 WiFi reconnected");
    wasConnected = true;
    wifiConnectedAt = millis();
    wifiReconnectAttempts = 0;
    phoneSyncedThisBoot = false;
    
    if (WiFi.hostByName("firestore.googleapis.com", firestoreIP)) {
      firestoreIPCached = true;
      Serial.print("🔒 Firestore IP cached: ");
      Serial.println(firestoreIP);
    }
  }

  if (!tcpipReady &&
      wifiConnectedAt != 0 &&
      millis() - wifiConnectedAt > WIFI_SETTLE_TIME) {

    tcpipReady = true;
    Serial.println("✅ TCP/IP stack stabilized");

    updateDeviceDoc(false, "none");
    lastSyncSend = millis();
  }

  if (tcpipReady && !phoneSyncedThisBoot) {
    pollCloud();
    phoneSyncedThisBoot = true;

    Serial.println("📲 Phone synced from device doc");
    Serial.print("📞 Owner phone: ");
    Serial.println(ownerPhone);
    Serial.print("🚑 Emergency phone: ");
    Serial.println(emergencyPhone);
  }

  // FIXED: Use full URL only, no IP overload
  if (tcpipReady && waitingCloudClear) {
    if (!canPerformHttpRequest()) return;
    
    markHttpRequestStart();
    WiFiClientSecure client;
    setupSecureClient(client);
    
    HTTPClient check;
    check.useHTTP10(true);
    
    // FIXED: Always use full URL string
    check.begin(client, deviceURL);
    
    check.setTimeout(4000);
    check.setConnectTimeout(5000);
    
    int code = check.GET();

    if (code == 200) {
      // UPGRADE 6: Reserve memory before storing HTTP response
      String body;
      body.reserve(512);
      body = check.getString();
      
      bool resetStillTrue = false;
      int r = body.indexOf("wifiReset");

      if (r != -1) {
        int rb = body.indexOf("booleanValue", r);
        if (rb != -1 && body.indexOf("true", rb) != -1) {
          resetStillTrue = true;
        }
      }

      if (!resetStillTrue) {
        waitingCloudClear = false;
        Serial.println("🔓 Cloud clean — normal mode");
      } else {
        Serial.println("⏳ Waiting cloud reset flag...");
      }
    }
    safeHttpEnd(check);
  }

  if (tcpipReady && millis() - lastSyncSend > HEARTBEAT_INTERVAL_MS) {
    bool fallActive =
      (fallState == ALERT_PENDING || fallState == POST_ALERT);

    const char* fallStatus =
      (fallState == ALERT_PENDING) ? "pending" :
      (fallState == POST_ALERT)    ? "sent" :
                                     "none";

    updateDeviceDoc(fallActive, fallStatus);
    lastSyncSend = millis();
  }

  if (tcpipReady &&
      !waitingCloudClear &&
      millis() - lastCloudCheck > POLL_INTERVAL_MS) {

    pollCloud();
  }
  
  if (alertCreationPending && millis() > alertCreationRetryTime) {
    if (WiFi.status() == WL_CONNECTED) {
      createAlert();
    } else {
      alertCreationRetryTime = millis() + ALERT_RETRY_INTERVAL;
    }
  }
  
  if (millis() - lastHeapCheck > 60000) {
    lastHeapCheck = millis();
    
    size_t freeHeap = ESP.getFreeHeap();
    size_t minFreeHeap = ESP.getMinFreeHeap();
    
    Serial.printf("💾 Heap: %d bytes (min: %d)\n", freeHeap, minFreeHeap);
    
    if (minFreeHeap < 10000) {
      Serial.println("🚨 CRITICAL: Low heap - forcing restart");
      ESP.restart();
    }
  }
}
