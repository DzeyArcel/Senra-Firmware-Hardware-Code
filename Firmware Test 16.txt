// ================= HELPER =================
void setWifiResetFlag(bool v) {
  EEPROM.write(WIFI_RESET_FLAG_ADDR, v ? 1 : 0);
  EEPROM.commit();
  delay(200);        // 🔒 give flash time
}


bool getWifiResetFlag() {
  return EEPROM.read(WIFI_RESET_FLAG_ADDR) == 1;
}


// ================= TIME =================
String isoTime() {
  time_t now;
  time(&now);
  struct tm* t = gmtime(&now);
  char buf[32];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", t);
  return String(buf);
}

// ================= EEPROM =================
void saveField(int addr, String v) {
  for (int i = 0; i < 32; i++)
    EEPROM.write(addr + i, i < v.length() ? v[i] : 0);
  EEPROM.commit();
}

String readField(int addr) {
  char b[33];
  for (int i = 0; i < 32; i++) b[i] = EEPROM.read(addr + i);
  b[32] = '\0';
  return String(b);
}

String readSSID() {
  return readField(SSID_ADDR);
}
String readPASS() {
  return readField(PASS_ADDR);
}
String readCaregiverId() {
  return readField(CAREGIVER_ADDR);
}

void handleSetWiFi() {
  Serial.println("📡 NEW WIFI RECEIVED");

  saveField(SSID_ADDR, server.arg("s"));
  saveField(PASS_ADDR, server.arg("p"));

  // 🔓 CLEAR EEPROM LOCK
  setWifiResetFlag(false);

  delay(300); // ⬅ REQUIRED

  // ✅ VERIFY FLAG
  if (getWifiResetFlag()) {
    Serial.println("❌ EEPROM CLEAR FAILED — ABORTING REBOOT");
    server.send(500, "text/plain", "EEPROM error, retry");
    return;
  }

  Serial.println("💾 EEPROM LOCK CLEARED");

  // ☁️ Clear cloud flag (failsafe)
  HTTPClient clear;
  clear.begin(deviceURL + "&updateMask.fieldPaths=wifiReset");
  clear.addHeader("Content-Type", "application/json");
  clear.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
  clear.end();

  delay(1500);

  Serial.println("🔁 REBOOT TO NORMAL MODE");
  ESP.restart();
}

// ================= WIFI =================
bool connectWiFi() {
  WiFi.mode(WIFI_STA);

  String ssid = readSSID();
  ssid.trim();
  String pass = readPASS();
  pass.trim();

  Serial.println("----- WIFI DEBUG -----");
  Serial.print("Stored SSID: [");
  Serial.print(ssid);
  Serial.println("]");
  Serial.print("Stored PASS length: ");
  Serial.println(pass.length());

  if (ssid.length() == 0) {
    Serial.println("No SSID found → Starting AP mode");
    startAP();
    return false;
  }

  Serial.println("Connecting to WiFi...");
  WiFi.begin(ssid.c_str(), pass.c_str());

  unsigned long t = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t < 20000) {
    delay(500);
    Serial.print(".");
  }
  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("✅ WIFI CONNECTED");
    Serial.print("IP: ");
    Serial.println(WiFi.localIP());

    // 🔥 MARK WIFI STABLE TIME
    wifiConnectedAt = millis();


    server.on("/setcaregiver", handleSetCaregiver);
    server.begin();
    return true;
  }

  Serial.println("❌ WIFI FAILED");
  Serial.print("WiFi status code: ");
  Serial.println(WiFi.status());

  startAP();
  return false;
}


// ================= GPS =================
void readGPS() {
  while (GPS_Serial.available()) gps.encode(GPS_Serial.read());
  if (gps.location.isValid()) {
    lastLat = gps.location.lat();
    lastLng = gps.location.lng();
  }
}

// ================= BATTERY =================
int readBatteryPercent() {
  float v = analogRead(BATTERY_PIN) / 4095.0 * 3.3 * 2.0;
  return constrain(map(v * 100, 330, 420, 0, 100), 0, 100);
}

  json += "} }";

 String url = deviceURL +
  "&updateMask.fieldPaths=status" +

http.addHeader("Content-Type", "application/json");

http.end();
}

// ================= CLOUD SAFETY =================
void handleUnlink(String pairedTo) {
  if (pairedTo.length() == 0 && readCaregiverId().length() > 0) {
    saveField(CAREGIVER_ADDR, "");
    fallState = IDLE;
    digitalWrite(BUZZER_PIN, LOW);

void handleWifiReset() {
  Serial.println("🔥 HARD WIFI RESET → REBOOT TO AP");

  saveField(SSID_ADDR, "");
  saveField(PASS_ADDR, "");

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(1500);

  ESP.restart();  // ✅ CLEAN REBOOT
}


void pollCloud() {
  if (!tcpipReady) return;

  // 🔒 Already rebooted for WiFi reset this boot
  if (wifiResetHandledThisBoot) return;

  // ⛔ TEMPORARY BLOCK DURING FALL / ALERT
  if (fallState == ALERT_PENDING || fallState == POST_ALERT) {
    Serial.println("⛔ WiFi reset blocked (active fall) — monitoring continues");
    return; // ✅ NO reboot, NO flags set
  }

  // 🔒 Wi-Fi must be connected
  if (WiFi.status() != WL_CONNECTED) return;

  // 🔒 Wi-Fi must be stable
  if (wifiConnectedAt == 0) return;
  if (millis() - wifiConnectedAt < WIFI_SETTLE_TIME) return;

  // 🔒 EEPROM lock means reset already consumed
  if (getWifiResetFlag()) return;

  HTTPClient http;
  http.begin(deviceURL);
  int code = http.GET();
  if (code != 200) {
    http.end();
    return;
  }

  String body = http.getString();
  http.end();

  if (body.indexOf("\"wifiReset\"") == -1) return;
  if (body.indexOf("true") == -1) return;

  Serial.println("🔥 CLOUD WIFI RESET CONFIRMED");

  // 🔒 Mark handled ONLY NOW
  wifiResetHandledThisBoot = true;

  // ✅ Clear cloud flag first
  HTTPClient clear;
  clear.begin(deviceURL + "&updateMask.fieldPaths=wifiReset");
  clear.addHeader("Content-Type", "application/json");
  clear.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
  clear.end();

  Serial.println("☁️ CLOUD FLAG CLEARED");

  delay(300);

  saveField(SSID_ADDR, "");
  saveField(PASS_ADDR, "");
  setWifiResetFlag(true);

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2000);

  Serial.println("🔁 REBOOTING TO AP MODE");
  ESP.restart();
}


// ================= FALL ENGINE =================
void checkFall() {
  static unsigned long postAlertAt = 0;

  if (fallState == POST_ALERT) {
    if (postAlertAt == 0) postAlertAt = millis();
    if (millis() - postAlertAt > 15000) {
      postAlertAt = 0;
      fallState = IDLE;
    }
    return;
  }

  sensors_event_t a, g, t;
  mpu.getEvent(&a, &g, &t);

  float totalG = sqrt(
    sq(a.acceleration.x / 9.81) +
    sq(a.acceleration.y / 9.81) +
    sq(a.acceleration.z / 9.81));

  unsigned long now = millis();

  if (fallState == IDLE && (totalG > IMPACT_G || abs(g.gyro.x) > ROT_SPIKE)) {
    fallState = CANDIDATE;
    fallTime = now;
  }

  if (fallState == CANDIDATE && now - fallTime > CONFIRM_DELAY) {
    if (totalG < STILL_G) {
      fallState = ALERT_PENDING;
      alertStart = now;
      updateDeviceDoc(true, "pending");
    } else {
      fallState = IDLE;
    }
  }

  if (fallState == ALERT_PENDING) {
    if (now - lastBuzz > BUZZ_INTERVAL) {
      digitalWrite(BUZZER_PIN, !digitalRead(BUZZER_PIN));
      lastBuzz = now;
    }

    if (digitalRead(BUTTON_PIN) == LOW) {
      digitalWrite(BUZZER_PIN, LOW);
      updateDeviceDoc(false, "cancelled");
      fallState = IDLE;
    }

    if (now - alertStart > CANCEL_WINDOW) {
      digitalWrite(BUZZER_PIN, LOW);
      createAlert();
      fallState = POST_ALERT;
    }
  }
}


// ================= SETUP =================
void setup() {
    tcpipReady = false;   // ✅ ADD THIS
  wifiConnectedAt = 0;  
  wifiResetHandledThisBoot = false;

// ✅ ADD THIS

  Serial.begin(115200);
  delay(1200);
  EEPROM.begin(EEPROM_SIZE);

  Serial.println("\n=== SENRA HARD BOOT ===");

  // HARD KILL RADIO FIRST
  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2500);

  // 🔒 ABSOLUTE AP LOCK
  if (getWifiResetFlag()) {
    Serial.println("🔥 WIFI RESET FLAG FOUND — FORCING AP");

    WiFi.persistent(false);
    WiFi.setAutoReconnect(false);
    WiFi.setSleep(false);

    WiFi.mode(WIFI_AP);
    bool ok = WiFi.softAP("SENRA_SETUP");

    Serial.print("AP START: ");
    Serial.println(ok ? "OK" : "FAILED");
    Serial.print("AP IP: ");
    Serial.println(WiFi.softAPIP());

    server.on("/", handleRoot);
    server.on("/setwifi", handleSetWiFi);
    server.begin();

    // HARD LOCK
    while (true) {
      server.handleClient();
      delay(10);
    }
  }

  // NORMAL BOOT
  Serial.println("Normal boot...");

  WiFi.persistent(false);
  WiFi.setAutoReconnect(true);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(BATTERY_PIN, INPUT);

  Wire.begin(21, 22);
  mpu.begin();
  

  GPS_Serial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  SIM800.begin(9600, SERIAL_8N1, SIM_RX, SIM_TX);

  configTime(0, 0, "pool.ntp.org", "time.google.com");

  connectWiFi();
}




// ================= LOOP =================
void loop() {
  server.handleClient();
  readGPS();
  checkFall();

  // 🔒 MARK TCP/IP READY ONLY ONCE
  if (WiFi.status() == WL_CONNECTED &&
      !tcpipReady &&
      wifiConnectedAt != 0 &&
      millis() - wifiConnectedAt > WIFI_SETTLE_TIME) {

    tcpipReady = true;
    Serial.println("✅ TCP/IP stack stabilized");

    // 🔥 CRITICAL: IMMEDIATE ONLINE SYNC
    Serial.println("☁️ Sending FIRST online sync");
    updateDeviceDoc(false, "none");
    lastSyncSend = millis();
  }

  // 🔁 NORMAL PERIODIC SYNC
  if (tcpipReady && millis() - lastSyncSend > 20000) {
    bool fallActive =
        (fallState == ALERT_PENDING || fallState == POST_ALERT);

    String fallStatus =
        (fallState == ALERT_PENDING) ? "pending" :
        (fallState == POST_ALERT)    ? "sent" :
                                       "none";

    updateDeviceDoc(fallActive, fallStatus);
    lastSyncSend = millis();
  }

  // ☁️ CLOUD COMMANDS
  if (tcpipReady && millis() - lastCloudCheck > 5000) {
    pollCloud();
    lastCloudCheck = millis();
  }
}
