#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <TinyGPSPlus.h>

// -------------------------------
// USER CONFIG
// -------------------------------
const char* WIFI_SSID     = "YOUR_WIFI";
const char* WIFI_PASSWORD = "YOUR_PASSWORD";

String DEVICE_ID = "SNR-TEST-04";
String PROJECT_ID = "senra-805d0";
String API_KEY    = "YOUR_API_KEY";

// -------------------------------
// FIRESTORE URL
// -------------------------------
String FIRESTORE_URL =
  "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID +
  "/databases/(default)/documents/devices/" + DEVICE_ID +
  "?key=" + API_KEY;

// -------------------------------
// HARDWARE OBJECTS
// -------------------------------
Adafruit_MPU6050 mpu;
TinyGPSPlus gps;
HardwareSerial GPS_Serial(2);

// GPS Pins
#define GPS_RX 16
#define GPS_TX 17

// Timing
unsigned long lastHeartbeat = 0;
const unsigned long HEARTBEAT_INTERVAL = 15000;

// Last GPS
double lastLat = 0;
double lastLng = 0;

// -------------------------------
// CONNECT WIFI
// -------------------------------
void connectWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  Serial.print("Connecting to WiFi");

  int tries = 0;
  while (WiFi.status() != WL_CONNECTED && tries < 20) {
    Serial.print(".");
    delay(500);
    tries++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi Connected!");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("\nWiFi FAILED.");
  }
}

// -------------------------------
// TIMESTAMP
// -------------------------------
String getISOTime() {
  time_t now;
  time(&now);
  struct tm *t = gmtime(&now);
  char buf[30];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", t);
  return String(buf);
}

// -------------------------------
// READ GPS
// -------------------------------
void readGPS() {
  while (GPS_Serial.available()) gps.encode(GPS_Serial.read());

  if (gps.location.isValid()) {
    lastLat = gps.location.lat();
    lastLng = gps.location.lng();
  }
}

// -------------------------------
// SEND HEARTBEAT
// -------------------------------
void sendHeartbeat() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;

  String url = FIRESTORE_URL +
               "&updateMask.fieldPaths=status"
               "&updateMask.fieldPaths=battery"
               "&updateMask.fieldPaths=lat"
               "&updateMask.fieldPaths=lng"
               "&updateMask.fieldPaths=lastSync";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  int battery = random(70, 95); // placeholder

  String payload = "{ \"fields\": {";
  payload += "\"status\": {\"stringValue\": \"online\"},";
  payload += "\"battery\": {\"integerValue\": \"" + String(battery) + "\"},";

  if (lastLat != 0 && lastLng != 0) {
    payload += "\"lat\": {\"doubleValue\": " + String(lastLat, 6) + "},";
    payload += "\"lng\": {\"doubleValue\": " + String(lastLng, 6) + "},";
  }

  payload += "\"lastSync\": {\"timestampValue\": \"" + getISOTime() + "\"}";
  payload += "} }";

  Serial.println("\n--- HEARTBEAT ---");
  Serial.println(payload);

  http.PATCH(payload);
  http.end();
}

// -------------------------------
// SEND FALL ALERT
// -------------------------------
void sendFallAlert() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  http.begin(FIRESTORE_URL);
  http.addHeader("Content-Type", "application/json");

  String payload = "{ \"fields\": {";
  payload += "\"fallDetected\": {\"booleanValue\": true},";
  payload += "\"fallTimestamp\": {\"timestampValue\": \"" + getISOTime() + "\"},";

  if (lastLat != 0 && lastLng != 0) {
    payload += "\"lat\": {\"doubleValue\": " + String(lastLat, 6) + "},";
    payload += "\"lng\": {\"doubleValue\": " + String(lastLng, 6) + "},";
  }

  payload += "\"status\": {\"stringValue\": \"fall\"}";
  payload += "} }";

  Serial.println("\n=== FALL ALERT SENT ===");
  Serial.println(payload);

  http.PATCH(payload);
  http.end();
}

// -------------------------------
// SIMPLE FALL DETECTION
// -------------------------------
bool impactDetected = false;
unsigned long impactTime = 0;

void detectFall() {
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  float ax = a.acceleration.x / 9.81;
  float ay = a.acceleration.y / 9.81;
  float az = a.acceleration.z / 9.81;

  float gForce = sqrt(ax*ax + ay*ay + az*az);

  if (gForce > 2.8 && !impactDetected) {
    impactDetected = true;
    impactTime = millis();
  }

  if (impactDetected && millis() - impactTime > 250) {
    if (gForce < 1.2) {
      sendFallAlert();
    }
    impactDetected = false;
  }
}

// -------------------------------
// SETUP
// -------------------------------
void setup() {
  Serial.begin(115200);

  WiFi.disconnect(true);
  delay(800);

  connectWiFi();
  configTime(0, 0, "pool.ntp.org");

  GPS_Serial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);

  if (!mpu.begin()) {
    Serial.println("MPU not found!");
  }
}

// -------------------------------
// LOOP
// -------------------------------
void loop() {
  readGPS();
  detectFall();

  if (millis() - lastHeartbeat > HEARTBEAT_INTERVAL) {
    sendHeartbeat();
    lastHeartbeat = millis();
  }
}
