// ================= HELPER =================
void setWifiResetFlag(bool v) {
  EEPROM.write(WIFI_RESET_FLAG_ADDR, v ? 1 : 0);
  EEPROM.commit();
  delay(200);
}

bool getWifiResetFlag() {
  return EEPROM.read(WIFI_RESET_FLAG_ADDR) == 1;
}

// ================= TIME =================
String isoTime() {
  time_t now;
  time(&now);
  struct tm* t = gmtime(&now);
  char buf[32];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", t);
  return String(buf);
}

// ================= EEPROM =================
void saveField(int addr, String v) {
  for (int i = 0; i < 32; i++)
    EEPROM.write(addr + i, i < v.length() ? v[i] : 0);
  EEPROM.commit();
}

String readField(int addr) {
  char b[33];
  for (int i = 0; i < 32; i++) b[i] = EEPROM.read(addr + i);
  b[32] = '\0';
  return String(b);
}

String readSSID() { return readField(SSID_ADDR); }
String readPASS() { return readField(PASS_ADDR); }
String readCaregiverId() { return readField(CAREGIVER_ADDR); }

// ================= AP MODE =================
void handleRoot() {
  server.send(200, "text/html",
              "<html><body><h2>Senra WiFi Setup</h2>"
              "<form action='/setwifi'>"
              "<input name='s'><input name='p' type='password'>"
              "<button>Save</button></form></body></html>");
}

void handleSetWiFi() {
  saveField(SSID_ADDR, server.arg("s"));
  saveField(PASS_ADDR, server.arg("p"));
  setWifiResetFlag(false);

  HTTPClient clear;
  clear.begin(deviceURL + "&updateMask.fieldPaths=wifiReset");
  clear.addHeader("Content-Type", "application/json");
  clear.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
  clear.end();

  delay(1500);
  ESP.restart();
}

void handleSetCaregiver() {
  if (!server.hasArg("id")) return;
  saveField(CAREGIVER_ADDR, server.arg("id"));
  server.send(200, "text/plain", "Caregiver linked");
}

void startAP() {
  WiFi.mode(WIFI_AP);
  WiFi.softAP("SENRA_SETUP");
  server.on("/", handleRoot);
  server.on("/setwifi", handleSetWiFi);
  server.on("/setcaregiver", handleSetCaregiver);
  server.begin();
}

// ================= WIFI =================
bool connectWiFi() {
  WiFi.mode(WIFI_STA);
  String ssid = readSSID(); ssid.trim();
  String pass = readPASS(); pass.trim();

  if (ssid.length() == 0) {
    startAP();
    return false;
  }

  WiFi.begin(ssid.c_str(), pass.c_str());
  unsigned long t = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t < 20000) delay(500);

  if (WiFi.status() == WL_CONNECTED) {
    wifiConnectedAt = millis();
    server.on("/setcaregiver", handleSetCaregiver);
    server.begin();
    return true;
  }

  startAP();
  return false;
}

// ================= GPS =================
void readGPS() {
  while (GPS_Serial.available()) gps.encode(GPS_Serial.read());
  if (gps.location.isValid()) {
    lastLat = gps.location.lat();
    lastLng = gps.location.lng();
  }
}

// ================= BATTERY =================
int readBatteryPercent() {
  float v = analogRead(BATTERY_PIN) / 4095.0 * 3.3 * 2.0;
  return constrain(map(v * 100, 330, 420, 0, 100), 0, 100);
}

// ================= FIRESTORE WRITE =================
void updateDeviceDoc(bool fall, String status) {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String json = "{ \"fields\": {";
  json += "\"status\":{\"stringValue\":\"online\"},";
  json += "\"fallDetected\":{\"booleanValue\":" + String(fall ? "true" : "false") + "},";
  json += "\"fallStatus\":{\"stringValue\":\"" + status + "\"},";
  json += "\"lat\":{\"doubleValue\":" + String(lastLat, 6) + "},";
  json += "\"lng\":{\"doubleValue\":" + String(lastLng, 6) + "},";
  json += "\"batteryLevel\":{\"integerValue\":\"" + String(readBatteryPercent()) + "\"},";
  json += "\"lastSync\":{\"timestampValue\":\"" + isoTime() + "\"}";
  json += "} }";

  String url = deviceURL +
               "&updateMask.fieldPaths=status"
               "&updateMask.fieldPaths=fallDetected"
               "&updateMask.fieldPaths=fallStatus"
               "&updateMask.fieldPaths=lat"
               "&updateMask.fieldPaths=lng"
               "&updateMask.fieldPaths=batteryLevel"
               "&updateMask.fieldPaths=lastSync";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  http.PATCH(json);
  http.end();
}

// ================= ALERT =================
void createAlert() {
  if (WiFi.status() != WL_CONNECTED) return;
  String caregiverId = readCaregiverId();
  if (caregiverId.length() == 0) return;

  HTTPClient http;
  String url = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID +
               "/databases/(default)/documents/alerts?key=" + API_KEY;

  String json = "{ \"fields\": {";
  json += "\"deviceId\":{\"stringValue\":\"" + DEVICE_ID + "\"},";
  json += "\"caregiverId\":{\"stringValue\":\"" + caregiverId + "\"},";
  json += "\"lat\":{\"doubleValue\":" + String(lastLat, 6) + "},";
  json += "\"lng\":{\"doubleValue\":" + String(lastLng, 6) + "},";
  json += "\"status\":{\"stringValue\":\"pending\"},";
  json += "\"timestamp\":{\"timestampValue\":\"" + isoTime() + "\"}";
  json += "} }";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  http.POST(json);
  http.end();
}

// ================= CLOUD SAFETY =================
void handleUnlink(String pairedTo) {
  if (pairedTo.length() == 0 && readCaregiverId().length() > 0) {
    saveField(CAREGIVER_ADDR, "");
    fallState = IDLE;
    digitalWrite(BUZZER_PIN, LOW);
  }
}

void handleWifiReset() {
  saveField(SSID_ADDR, "");
  saveField(PASS_ADDR, "");
  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(1500);
  ESP.restart();
}

void pollCloud() {
  if (!tcpipReady || wifiResetHandledThisBoot) return;
  if (fallState == ALERT_PENDING || fallState == POST_ALERT) return;
  if (WiFi.status() != WL_CONNECTED) return;
  if (wifiConnectedAt == 0 || millis() - wifiConnectedAt < WIFI_SETTLE_TIME) return;
  if (getWifiResetFlag()) return;

  HTTPClient http;
  http.begin(deviceURL);
  if (http.GET() != 200) { http.end(); return; }

  String body = http.getString();
  http.end();

  if (body.indexOf("\"wifiReset\"") == -1 || body.indexOf("true") == -1) return;

  wifiResetHandledThisBoot = true;

  HTTPClient clear;
  clear.begin(deviceURL + "&updateMask.fieldPaths=wifiReset");
  clear.addHeader("Content-Type", "application/json");
  clear.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
  clear.end();

  saveField(SSID_ADDR, "");
  saveField(PASS_ADDR, "");
  setWifiResetFlag(true);

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2000);
  ESP.restart();
}
// ================= ORIENTATION FILTER =================
void updateOrientation(float ax, float ay, float az, float gx, float gy) {
  unsigned long nowMicros = micros();
  if (lastMicros == 0) lastMicros = nowMicros;
  float dt = (nowMicros - lastMicros) / 1e6f;
  lastMicros = nowMicros;

  float accelRoll  = atan2(ay, az) * 57.2958;
  float accelPitch = atan2(-ax, sqrt(ay * ay + az * az)) * 57.2958;

  static float rollFilt = 0, pitchFilt = 0;
  rollFilt  = 0.98 * (rollFilt  + gx * 57.2958 * dt) + 0.02 * accelRoll;
  pitchFilt = 0.98 * (pitchFilt + gy * 57.2958 * dt) + 0.02 * accelPitch;

  rollDeg = rollFilt;
  pitchDeg = pitchFilt;
}

// ================= FALL ENGINE =================
void checkFall() {
  static unsigned long postAlertAt = 0;

  if (fallState == POST_ALERT) {
    if (postAlertAt == 0) postAlertAt = millis();
    if (millis() - postAlertAt > 15000) {
      postAlertAt = 0;
      fallState = IDLE;
    }
    return;
  }

  sensors_event_t a, g, t;
  mpu.getEvent(&a, &g, &t);

  float ax = a.acceleration.x;
  float ay = a.acceleration.y;
  float az = a.acceleration.z;

  float ax_g = ax / 9.81;
  float ay_g = ay / 9.81;
  float az_g = az / 9.81;

  float accelMag_g = sqrt(ax_g * ax_g + ay_g * ay_g + az_g * az_g);

  float gx_dps = g.gyro.x * 57.2958;
  float gy_dps = g.gyro.y * 57.2958;
  float gz_dps = g.gyro.z * 57.2958;
  float gyroMag_dps = sqrt(gx_dps * gx_dps + gy_dps * gy_dps + gz_dps * gz_dps);

  updateOrientation(ax, ay, az, g.gyro.x, g.gyro.y);

  postureChanged = abs(pitchDeg - lastPitchDeg) > 35;
  lastPitchDeg = pitchDeg;

  bool normalImpact  = accelMag_g > IMPACT1_G;
  bool strongImpact  = accelMag_g > IMPACT2_G;
  bool rotationSpike = gyroMag_dps > ROT_SPIKE;

  bool slowFall =
      abs(pitchDeg) > SLOW_FALL_TILT_DEG &&
      gyroMag_dps > SLOW_FALL_GYRO_DPS &&
      accelMag_g < SLOW_FALL_MAX_G;

  unsigned long now = millis();

  if (fallState == IDLE &&
      (strongImpact || (normalImpact && rotationSpike) || slowFall)) {

    fallState = CANDIDATE;
    firstImpactMs = now;
    lastImpactMs = now;
    stillStartMs = 0;
    return;
  }

  if (fallState == CANDIDATE) {
    if (now - lastImpactMs < GRACE_AFTER_IMPACT_MS) return;

    bool accelStill = accelMag_g > STILL_MIN_G && accelMag_g < STILL_MAX_G;
    bool gyroStill  = gyroMag_dps < STILL_GYRO_MAX;

    if (postureChanged && accelStill && gyroStill) {
      if (stillStartMs == 0) stillStartMs = now;

      if (now - stillStartMs >= STILL_CONFIRM_MS) {
        fallState = ALERT_PENDING;
        alertStart = now;
        updateDeviceDoc(true, "pending");
      }
    } else {
      stillStartMs = 0;
    }

    if (now - firstImpactMs > FALL_WINDOW_MS && stillStartMs == 0) {
      fallState = IDLE;
    }
  }

  if (fallState == ALERT_PENDING) {
    if (now - lastBuzz > BUZZ_INTERVAL) {
      digitalWrite(BUZZER_PIN, !digitalRead(BUZZER_PIN));
      lastBuzz = now;
    }

    if (digitalRead(BUTTON_PIN) == LOW) {
      digitalWrite(BUZZER_PIN, LOW);
      updateDeviceDoc(false, "cancelled");
      fallState = IDLE;
    }

    if (now - alertStart > CANCEL_WINDOW) {
      digitalWrite(BUZZER_PIN, LOW);
      createAlert();
      fallState = POST_ALERT;
    }
  }
}

// ================= SETUP =================
void setup() {
  tcpipReady = false;
  wifiConnectedAt = 0;
  wifiResetHandledThisBoot = false;

  Serial.begin(115200);
  delay(1200);
  EEPROM.begin(EEPROM_SIZE);

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2500);

  if (getWifiResetFlag()) {
    WiFi.mode(WIFI_AP);
    WiFi.softAP("SENRA_SETUP");
    server.on("/", handleRoot);
    server.on("/setwifi", handleSetWiFi);
    server.begin();
    while (true) {
      server.handleClient();
      delay(10);
    }
  }

  WiFi.persistent(false);
  WiFi.setAutoReconnect(true);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(BATTERY_PIN, INPUT);

  Wire.begin(21, 22);
  mpu.begin();

  GPS_Serial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  SIM800.begin(9600, SERIAL_8N1, SIM_RX, SIM_TX);

  configTime(0, 0, "pool.ntp.org", "time.google.com");

  connectWiFi();
}

// ================= LOOP =================
void loop() {
  server.handleClient();
  readGPS();
  checkFall();

  if (WiFi.status() == WL_CONNECTED &&
      !tcpipReady &&
      wifiConnectedAt != 0 &&
      millis() - wifiConnectedAt > WIFI_SETTLE_TIME) {

    tcpipReady = true;
    updateDeviceDoc(false, "none");
    lastSyncSend = millis();
  }

  if (tcpipReady && millis() - lastSyncSend > 20000) {
    bool fallActive =
        (fallState == ALERT_PENDING || fallState == POST_ALERT);

    String fallStatus =
        (fallState == ALERT_PENDING) ? "pending" :
        (fallState == POST_ALERT)    ? "sent" :
                                       "none";

    updateDeviceDoc(fallActive, fallStatus);
    lastSyncSend = millis();
  }

  if (tcpipReady && millis() - lastCloudCheck > 5000) {
    pollCloud();
    lastCloudCheck = millis();
  }
}
