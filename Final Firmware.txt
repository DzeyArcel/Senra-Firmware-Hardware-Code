void saveField(int addr, String v) {
  for (int i = 0; i < 32; i++) {
    EEPROM.write(addr + i, i < v.length() ? v[i] : 0);
  }
  EEPROM.commit();
}

String readField(int addr) {
  char b[33];
  for (int i = 0; i < 32; i++) {
    b[i] = EEPROM.read(addr + i);
  }
  b[32] = '\0';
  return String(b);
}

const char* isoTime() {
  static char buf[32];
  time_t now;
  time(&now);
  struct tm* t = gmtime(&now);
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", t);
  return buf;
}

String extractStringField(const String& json, const String& field) {
  int p = json.indexOf("\"" + field + "\"");
  if (p == -1) return "";

  int v = json.indexOf("stringValue", p);
  if (v == -1) return "";

  int q1 = json.indexOf("\"", v + 12);
  int q2 = json.indexOf("\"", q1 + 1);
  if (q1 == -1 || q2 == -1) return "";

  return json.substring(q1 + 1, q2);
}

bool canPerformHttpRequest() {
  if (httpRequestInProgress && millis() - httpStartTime > 30000) {

}

void markHttpRequestStart() {
  httpRequestInProgress = true;

void markHttpRequestEnd() {
  httpRequestInProgress = false;
}

void safeHttpEnd(HTTPClient& http) {
  http.end();
  markHttpRequestEnd();
  yield();
}

int readBatteryPercent() {
  const int samples = 25;
  float total = 0;

  for (int i = 0; i < samples; i++) {
    total += analogRead(BATTERY_PIN);
    delayMicroseconds(300);
  }

  float raw = total / samples;
  float voltage = raw * 3.3 / 4095.0 * 2.0;
  static float smoothVoltage = voltage;
  smoothVoltage = smoothVoltage * 0.90 + voltage * 0.10;
  float corrected = smoothVoltage + 0.58;

  if (corrected > 4.2) corrected = 4.2;

  int percent;
  if (corrected >= 4.15) percent = 100;
  else if (corrected >= 4.05) percent = 95;
  else if (corrected >= 4.00) percent = 85;
  else if (corrected >= 3.90) percent = 75;
  else if (corrected >= 3.80) percent = 65;
  else if (corrected >= 3.70) percent = 55;
  else if (corrected >= 3.60) percent = 45;
  else if (corrected >= 3.50) percent = 35;
  else if (corrected >= 3.40) percent = 25;
  else if (corrected >= 3.30) percent = 15;
  else percent = 5;

  static int lastPercent = percent;
  lastPercent = (lastPercent * 3 + percent) / 4;

  return lastPercent;
}

void readGPS() {
  static unsigned long lastGPSUpdate = 0;
  if (millis() - lastGPSUpdate < 500) return;
  lastGPSUpdate = millis();

  int processed = 0;
  while (GPS_Serial.available() && processed < 32) {
    if (gps.encode(GPS_Serial.read())) {
      if (gps.location.isValid() && gps.location.age() < 2000) {
        noInterrupts();
        lastLat = gps.location.lat();
        lastLng = gps.location.lng();
        interrupts();
      }
    }
    processed++;
  }
}

bool simReady() {

  while (SIM800.available()) SIM800.read();

  SIM800.println("AT");

  unsigned long start = millis();
  String resp = "";

  while (millis() - start < 1500) {
    if (SIM800.available()) {
      resp += char(SIM800.read());
    }
  }

  if (resp.indexOf("OK") != -1) {
    return true;
  }

  return false;
}

void checkSIMRecovery() {
  if (WiFi.status() == WL_CONNECTED) return;
  static unsigned long lastSimCheck = 0;
  if (millis() - lastSimCheck < 120000) return;

  lastSimCheck = millis();
  bool ready = simReady();
  bool registered = false;
  if (ready) registered = simRegistered();

  if (!ready || !registered) {
    simInitialized = false;
    initSIM();
  }
}

bool sendSMS_Safe(const String& number, const String& message) {

  if (number.length() < 10) return false;

  while (SIM800.available()) SIM800.read();

  SIM800.println("AT+CMGF=1");
  delay(300);

  SIM800.print("AT+CMGS=\"");
  SIM800.print(number);
  SIM800.println("\"");

  unsigned long start = millis();
  String resp = "";

  // wait for prompt
  while (millis() - start < 5000) {
    if (SIM800.available()) {
      resp += char(SIM800.read());
      if (resp.indexOf(">") != -1) break;
    }
  }

  if (resp.indexOf(">") == -1) {
    return false;
  }

  SIM800.print(message);
  SIM800.write(26); // CTRL+Z

  start = millis();
  resp = "";

  while (millis() - start < 10000) {
    if (SIM800.available()) {
      resp += char(SIM800.read());

      if (resp.indexOf("+CMGS") != -1 || resp.indexOf("OK") != -1) {
        return true;
      }

      if (resp.indexOf("ERROR") != -1) {
  
        return false;
      }
    }
  }

  
  return false;
}

void processSMSQueue() {

  SMSQueueItem &sms = smsRetryQueue[0];

  if (!sms.pending) return;
  if (smsInProgress) return;

  // allow sending even if WiFi reconnects
  if (WiFi.status() == WL_CONNECTED && tcpipReady) return;

 

  // check retry limit
  if (sms.retries >= MAX_SMS_RETRIES) {
    

    sms.pending = false;

    if (fallState == ALERT_PENDING || fallState == POST_ALERT) {
      lastSyncSend = millis();
      updateDeviceDoc(true, "sms_failed");
    }

    return;
  }

  // ensure SIM is ready
  if (!simReady()) {
    
    initSIM();
    delay(2000);
    return;
  }

  if (!simRegistered()) {
   
    delay(2000);
    return;
  }

  

  smsInProgress = true;

  bool sent = sendSMS_Safe(sms.number, sms.message);

  smsInProgress = false;

  if (sent) {

    

    sms.pending = false;

    if (fallState == ALERT_PENDING || fallState == POST_ALERT) {
      lastSyncSend = millis();
      updateDeviceDoc(true, "sent");
    }

  } else {

    

    sms.retries++;
    sms.lastAttempt = millis();
  }
}

bool createAlert() {

  return false;

  String caregiverId = readCaregiverId();
  if (caregiverId.length() == 0) return false;

  alertLat = lastLat;
  alertLng = lastLng;
  interrupts();

  markHttpRequestStart();

  snprintf(json, sizeof(json), "{\"fields\":{\"deviceId\":{\"stringValue\":\"%s\"},\"caregiverId\":{\"stringValue\":\"%s\"},\"lat\":{\"doubleValue\":%.6f},\"lng\":{\"doubleValue\":%.6f},\"status\":{\"stringValue\":\"pending\"},\"timestamp\":{\"timestampValue\":\"%s\"},\"firmwareModel\":{\"stringValue\":\"%s\"},\"firmwareVersion\":{\"stringValue\":\"%s\"}}}", DEVICE_ID.c_str(), caregiverId.c_str(), alertLat, alertLng, isoTime(), FIRMWARE_MODEL, FIRMWARE_VERSION);

    alertAlreadySent = true;
    lastSyncSend = millis();
    safeHttpEnd(http);
    return true;
  }

  safeHttpEnd(http);
  return false;
}

void pollCloud() {
  if (!tcpipReady || wifiResetHandledThisBoot || fallState == ALERT_PENDING || fallState == POST_ALERT || WiFi.status() != WL_CONNECTED || wifiConnectedAt == 0 || millis() - wifiConnectedAt < WIFI_SETTLE_TIME || getWifiResetFlag() || !canPerformHttpRequest() || millis() - lastCloudCheck < POLL_INTERVAL) return;

  markHttpRequestStart();
  WiFiClientSecure client;
  client.setInsecure();
  client.setTimeout(8000);
  HTTPClient http;

  String url = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID + "/databases/(default)/documents/devices/" + DEVICE_ID + "?key=" + API_KEY;
  http.begin(client, url);
  http.setTimeout(HTTP_TIMEOUT);

  int code = http.GET();
  lastCloudCheck = millis();

  if (code != HTTP_CODE_OK) {
    safeHttpEnd(http);
    return;
  }

  String body = http.getString();
  safeHttpEnd(http);

  int op = body.indexOf("\"ownerPhone\"");
  if (op != -1) {
    int v = body.indexOf("stringValue", op);
    int q1 = body.indexOf("\"", v + 12);
    int q2 = body.indexOf("\"", q1 + 1);

    if (q1 != -1 && q2 != -1) {
      String newOwner = body.substring(q1 + 1, q2);
      newOwner.trim();
      if (newOwner.length() >= 10 && newOwner != ownerPhone) {
        ownerPhone = newOwner;
        saveField(PAIRED_PHONE_ADDR, ownerPhone);
      }
    }
  }

  int ep = body.indexOf("\"emergencyPhone\"");
  if (ep != -1) {
    int v = body.indexOf("stringValue", ep);
    int q1 = body.indexOf("\"", v + 12);
    int q2 = body.indexOf("\"", q1 + 1);

    if (q1 != -1 && q2 != -1) {
      String newEmergency = body.substring(q1 + 1, q2);
      newEmergency.trim();
      bool valid = newEmergency.length() >= 10;
      for (char c : newEmergency) {
        if (!(isDigit(c) || c == '+')) valid = false;
      }
      if (valid && newEmergency != emergencyPhone) {
        emergencyPhone = newEmergency;
        saveField(EMERGENCY_PHONE_ADDR, emergencyPhone);
      }
    }
  }

  int p = body.indexOf("\"paired_to\"");
  if (p != -1) {
    int v = body.indexOf("stringValue", p);
    if (v != -1) {
      int q1 = body.indexOf("\"", v + 12);
      int q2 = body.indexOf("\"", q1 + 1);
      if (q1 != -1 && q2 != -1 && q2 > q1) {
        String pairedTo = body.substring(q1 + 1, q2);
        pairedTo.trim();
        handleUnlink(pairedTo);
      }
    }
  }

  bool wifiResetRequested = false;
  bool wifiResetAcked = false;

  int r = body.indexOf("\"wifiReset\"");
  if (r != -1) {
    int rb = body.indexOf("booleanValue", r);
    if (rb != -1 && body.indexOf("true", rb) != -1) wifiResetRequested = true;
  }

  int a = body.indexOf("\"wifiResetAck\"");
  if (a != -1) {
    int ab = body.indexOf("booleanValue", a);
    if (ab != -1 && body.indexOf("true", ab) != -1) wifiResetAcked = true;
  }

  if (!wifiResetRequested || wifiResetAcked) return;

  markDeviceResetting();
  delay(800);
  wifiResetHandledThisBoot = true;

  if (canPerformHttpRequest()) {
    markHttpRequestStart();
    WiFiClientSecure ackClient;
    ackClient.setInsecure();
    ackClient.setTimeout(8000);
    HTTPClient ack;

    String ackUrl = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID + "/databases/(default)/documents/devices/" + DEVICE_ID + "?key=" + API_KEY + "&updateMask.fieldPaths=wifiResetAck";
    ack.begin(ackClient, ackUrl);
    ack.addHeader("Content-Type", "application/json");
    ack.setTimeout(HTTP_TIMEOUT);
    ack.PATCH("{\"fields\":{\"wifiResetAck\":{\"booleanValue\":true}}}");
    safeHttpEnd(ack);

    delay(100);

    markHttpRequestStart();
    WiFiClientSecure clearClient;
    clearClient.setInsecure();
    clearClient.setTimeout(8000);
    HTTPClient clear;

    String clearUrl = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID + "/databases/(default)/documents/devices/" + DEVICE_ID + "?key=" + API_KEY + "&updateMask.fieldPaths=wifiReset";
    clear.begin(clearClient, clearUrl);
    clear.addHeader("Content-Type", "application/json");
    clear.setTimeout(HTTP_TIMEOUT);
    clear.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
    safeHttpEnd(clear);
  }

  delay(300);
  saveField(SSID_ADDR, "");
  saveField(PASS_ADDR, "");
  setWifiResetFlag(true);

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2000);
  ESP.restart();
}

bool initMPU() {
  Wire.begin(21, 22);
  Wire.setClock(100000);

  if (!mpu.begin()) {
    mpuInitialized = false;
    return false;
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
  mpuInitialized = true;
  return true;
}

void checkFall() {

  esp_task_wdt_reset();

  static unsigned long postAlertAt = 0;
  static float filteredG = 1.0;
  static bool freeFallDetected = false;
  static unsigned long freeFallTime = 0;
  static bool buzzerState = false;

  unsigned long now = millis();

  if (now < fallDetectionStart) return;
  if (now < fallCooldownUntil) return;

  if (!mpuInitialized && (now - lastMpuCheck > 30000)) {
    lastMpuCheck = now;
    initMPU();
    return;
  }

  if (!mpuInitialized) return;

  sensors_event_t a, g, t;

  if (!mpu.getEvent(&a, &g, &t)) {
    mpuInitialized = false;
    return;
  }

  if (!mpuReady) {
    if (now - mpuInitTime > 800) {
      mpuReady = true;
    } else return;
  }

  // ================= POST ALERT =================

  if (fallState == POST_ALERT) {

    if (postAlertAt == 0) postAlertAt = now;

    if (now - postAlertAt > 20000) {

      digitalWrite(BUZZER_PIN, LOW);

      postAlertAt = 0;

      fallState = IDLE;

      freeFallDetected = false;
      filteredG = 1.0;

      alertAlreadySent = false;
      smsSentThisFall = false;
      smsInProgress = false;

      fallCooldownUntil = now + FALL_COOLDOWN_MS;

      if (WiFi.status() == WL_CONNECTED) {
        updateDeviceDoc(false, "cooldown");
      }
    }

    return;
  }

  // ================= ACCELERATION =================

  float totalG = sqrt(
    sq(a.acceleration.x / 9.81) +
    sq(a.acceleration.y / 9.81) +
    sq(a.acceleration.z / 9.81)
  );

  filteredG = 0.7 * filteredG + 0.3 * totalG;

  float rot = max(max(abs(g.gyro.x), abs(g.gyro.y)), abs(g.gyro.z));

  // ================= IDLE =================

  if (fallState == IDLE) {

if (!freeFallDetected && filteredG < FREE_FALL_G) {
  freeFallDetected = true;
  freeFallTime = now;
}

    if (freeFallDetected && (filteredG > IMPACT_G || rot > ROT_SPIKE)) {

      fallState = CANDIDATE;
      fallTime = now;

      freeFallDetected = false;
    }

   if (freeFallDetected) {

  if (now - freeFallTime > IMPACT_TIMEOUT) {
    freeFallDetected = false;
  }

  if (filteredG > 3.5) {
    freeFallDetected = false;
  }

}

    return;
  }

  // ================= CANDIDATE =================

  if (fallState == CANDIDATE) {

    if (now - fallTime > CONFIRM_DELAY) {

      if (filteredG < STILL_G && (now - fallTime) > STILL_CONFIRM) {

        fallState = ALERT_PENDING;
        alertStart = now;
        lastBuzz = 0;
        buzzerState = false;

        digitalWrite(BUZZER_PIN, HIGH);

        alertAlreadySent = false;

        if (WiFi.status() == WL_CONNECTED) {
          updateDeviceDoc(true, "pending");
        }

      } else {

        fallState = IDLE;
      }
    }

    return;
  }

  // ================= ALERT PENDING =================

  if (fallState == ALERT_PENDING) {

    if (now - lastBuzz > BUZZ_INTERVAL) {

      buzzerState = !buzzerState;
      digitalWrite(BUZZER_PIN, buzzerState);

      lastBuzz = now;
    }

    static unsigned long lastButtonCheck = 0;
    static int buttonStableCount = 0;

    if (now - lastButtonCheck > 50) {

      lastButtonCheck = now;

      if (digitalRead(BUTTON_PIN) == LOW) {

        buttonStableCount++;

        if (buttonStableCount >= 3) {

          cancelDisplayUntil = millis() + CANCEL_DISPLAY_TIME;

          digitalWrite(BUZZER_PIN, LOW);

          smsSentThisFall = false;
          alertAlreadySent = false;

          for (int i = 0; i < 2; i++) {
            smsRetryQueue[i].pending = false;
          }

          if (WiFi.status() == WL_CONNECTED) {
            updateDeviceDoc(false, "cancelled_by_device");
          }

          fallState = IDLE;

          buttonStableCount = 0;

          return;
        }

      } else {

        buttonStableCount = 0;
      }
    }

    // ================= SEND ALERT =================

    if (now - alertStart > CANCEL_WINDOW) {

      digitalWrite(BUZZER_PIN, LOW);

      if (!alertAlreadySent) {

        if (WiFi.status() == WL_CONNECTED) {

          if (createAlert()) {
            updateDeviceDoc(true, "sent");
          }

        } else if (!smsSentThisFall && emergencyPhone.length() > 10) {

          char msg[200];

          double lat;
          double lng;

          noInterrupts();
          lat = lastLat;
          lng = lastLng;
          interrupts();

          snprintf(
            msg,
            sizeof(msg),
            "SENRA ALERT\n%s FALL\n%.5f,%.5f\nmaps.google.com/?q=%.5f,%.5f",
            DEVICE_ID.c_str(),
            lat,
            lng,
            lat,
            lng
          );

          smsRetryQueue[0].pending = true;
          smsRetryQueue[0].number = emergencyPhone;
          smsRetryQueue[0].message = String(msg);

          smsSentThisFall = true;
        }

        alertAlreadySent = true;
      }

      fallState = POST_ALERT;
      postAlertAt = now;
    }
  }
}

void handleSetWiFi() {
  saveField(SSID_ADDR, server.arg("s"));
  saveField(PASS_ADDR, server.arg("p"));
  setWifiResetFlag(false);
  delay(300);

  if (hasCaregiverPaired()) {
    devicePaired = true;
  }

  if (getWifiResetFlag()) {
    server.send(500, "text/plain", "EEPROM error, retry");
    return;
  }

  if (WiFi.status() == WL_CONNECTED && canPerformHttpRequest()) {
    markHttpRequestStart();
    WiFiClientSecure client1;
    client1.setInsecure();
    client1.setTimeout(8000);
    HTTPClient clear1;

    String url1 = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID + "/databases/(default)/documents/devices/" + DEVICE_ID + "?key=" + API_KEY + "&updateMask.fieldPaths=wifiReset";
    clear1.begin(client1, url1);
    clear1.addHeader("Content-Type", "application/json");
    clear1.setTimeout(HTTP_TIMEOUT);
    clear1.PATCH("{\"fields\":{\"wifiReset\":{\"booleanValue\":false}}}");
    safeHttpEnd(clear1);

    delay(100);

    markHttpRequestStart();
    WiFiClientSecure client2;
    client2.setInsecure();
    client2.setTimeout(8000);
    HTTPClient clear2;

    String url2 = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID + "/databases/(default)/documents/devices/" + DEVICE_ID + "?key=" + API_KEY + "&updateMask.fieldPaths=wifiResetAck";
    clear2.begin(client2, url2);
    clear2.addHeader("Content-Type", "application/json");
    clear2.setTimeout(HTTP_TIMEOUT);
    clear2.PATCH("{\"fields\":{\"wifiResetAck\":{\"booleanValue\":false}}}");
    safeHttpEnd(clear2);
  }

  delay(1200);
  ESP.restart();
}

void handleSetCaregiver() {

void startAP() {
  WiFi.disconnect(true);
  WiFi.mode(WIFI_AP);
  WiFi.softAP("SENRA_SETUP");

  server.on("/", handleRoot);
  server.on("/setwifi", handleSetWiFi);
  server.on("/setcaregiver", handleSetCaregiver);
  server.begin();
}

bool connectWiFi() {

  if (apModeActive) return false;

  String ssid = readSSID();
  String pass = readPASS();

  if (ssid.length() == 0) {
    apModeActive = true;
    startAP();
    return false;
  }

  // --- HARD RESET WIFI DRIVER ---
  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
  delay(800);

  WiFi.mode(WIFI_STA);

  WiFi.setTxPower(WIFI_POWER_15dBm);
  WiFi.setSleep(false);
  WiFi.persistent(false);
  WiFi.setAutoReconnect(true);

  delay(100);

  WiFi.begin(ssid.c_str(), pass.c_str());

  unsigned long startAttempt = millis();

  // --- WAIT FOR CONNECTION ---
  while (WiFi.status() != WL_CONNECTED && millis() - startAttempt < 30000) {
    delay(500);
    yield();
  }

  if (WiFi.status() == WL_CONNECTED) {

    wifiConnectedAt = millis();
    apModeActive = false;

    // keep caregiver endpoint availablefk
    server.on("/setcaregiver", handleSetCaregiver);
    server.begin();

    return true;
  }

  // --- FAILED → RETURN TO AP MODE ---
  apModeActive = true;
  startAP();

  return false;
}
void setup() {
  Serial.begin(115200);
  delay(100);

  tcpipReady = false;
  wifiConnectedAt = 0;
  wifiResetHandledThisBoot = false;
  phoneSyncedThisBoot = false;
  bootTime = millis();

  delay(1200);
  EEPROM.begin(EEPROM_SIZE);

  ESP_ERROR_CHECK(esp_task_wdt_init(30, true));
  ESP_ERROR_CHECK(esp_task_wdt_add(NULL));

  devicePaired = hasCaregiverPaired();

  ownerPhone = readField(PAIRED_PHONE_ADDR);
  ownerPhone.trim();

  bool ownerValid = ownerPhone.length() >= 10;
  for (char c : ownerPhone) {
    if (!(isDigit(c) || c == '+')) ownerValid = false;
  }
  if (!ownerValid) {
    ownerPhone = "";
    saveField(PAIRED_PHONE_ADDR, "");
  }

  emergencyPhone = readField(EMERGENCY_PHONE_ADDR);
  emergencyPhone.trim(); 
  bool emergencyValid = emergencyPhone.length() >= 10;
  for (char c : emergencyPhone) {
    if (!(isDigit(c) || c == '+')) emergencyValid = false;
  }
  if (!emergencyValid) {
    emergencyPhone = "";
    saveField(EMERGENCY_PHONE_ADDR, "");
  }

  WiFi.disconnect(true, true);
  WiFi.mode(WIFI_OFF);
  delay(2500);

  if (getWifiResetFlag() || !hasWiFiCredentials()) {
    apModeActive = true;

    WiFi.persistent(false);
    WiFi.setAutoReconnect(false);
    WiFi.setSleep(false);
    WiFi.mode(WIFI_AP);
    WiFi.softAP("SENRA_SETUP");

    server.on("/", handleRoot);
    server.on("/setwifi", handleSetWiFi);
    server.on("/setcaregiver", handleSetCaregiver);
    server.begin();
  } else {
    apModeActive = false;
  }

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(BATTERY_PIN, INPUT); 
  analogSetPinAttenuation(BATTERY_PIN, ADC_11db);
  analogReadResolution(12);

  initMPU();
  mpuInitTime = millis();
  mpuReady = false;  

  fallDetectionStart = millis() + 5000;

  GPS_Serial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  SIM800.begin(115200, SERIAL_8N1, SIM_RX, SIM_TX);
  initSIM();

  configTime(0, 0, "pool.ntp.org", "time.google.com");

  if (!apModeActive) {
    connectWiFi();
  }

  SIM800.println("AT+CSQ");
  delay(1000);
  while(SIM800.available()){
    Serial.write(SIM800.read());
  }
}

void loop() {
  esp_task_wdt_reset();
  readGPS();
  server.handleClient();

  if (devicePaired) {
    esp_task_wdt_reset();
    checkFall();

    esp_task_wdt_reset();
    processSMSQueue();

    esp_task_wdt_reset();
    checkSIMRecovery();
  }


  if (millis() - lastBatteryRead > 60000) {
    batteryLevel = readBatteryPercent();
    lastBatteryRead = millis();
  }

  if (!apModeActive) {
    static bool wasConnected = false;

    if (WiFi.status() != WL_CONNECTED) {
      if (wasConnected) {
        markDeviceOffline();
        wasConnected = false;
        tcpipReady = false;
        httpRequestInProgress = false;
      }
      delay(100);
      return;
    }

    if (!wasConnected) {
      wasConnected = true;
      wifiConnectedAt = millis();
      phoneSyncedThisBoot = false;
    }

    if (!tcpipReady && wifiConnectedAt != 0 && (millis() - wifiConnectedAt > WIFI_SETTLE_TIME)) {
      tcpipReady = true;
      updateDeviceDoc(false, "none");
      lastSyncSend = millis();
    }

    if (tcpipReady && !phoneSyncedThisBoot) {
      pollCloud();
      phoneSyncedThisBoot = true;
    }

    if (tcpipReady && (millis() - lastSyncSend > HEARTBEAT_INTERVAL)) {
      const char* fallStatus = "none";
      bool fallActive = (fallState == ALERT_PENDING || fallState == POST_ALERT);

      if (fallState == ALERT_PENDING) fallStatus = "pending";
      else if (fallState == POST_ALERT) fallStatus = "sent";
      else if (millis() < cancelDisplayUntil)fallStatus = "cancelled_by_device";

      updateDeviceDoc(fallActive, fallStatus);
      lastSyncSend = millis();
    }

    if (tcpipReady && (millis() - lastCloudCheck > POLL_INTERVAL)) {
      pollCloud();
    }
  }

  if (millis() - bootTime > DAY_MILLIS) {
    ESP.restart();
  }

  delay(5);
}
