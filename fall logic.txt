#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <math.h>

Adafruit_MPU6050 mpu;

// ================== TUNABLE CONSTANTS ==================

// Impact thresholds (in g)
const float IMPACT1_G = 2.0;     // normal impact to start monitoring
const float IMPACT2_G = 2.7;     // strong impact (often a real fall)

// Rotation threshold (deg/s)
const float GYRO_SPIKE_DPS = 130.0;

// Stillness thresholds
const float STILL_MIN_G = 0.7;   // near 1g but with noise
const float STILL_MAX_G = 1.3;
const float STILL_GYRO_MAX = 40.0;

// Timings (ms)
const unsigned long GRACE_AFTER_IMPACT_MS = 400;   // wait after impact before checking stillness
const unsigned long STILL_CONFIRM_MS      = 1800;  // must be still this long to confirm fall
const unsigned long FALL_WINDOW_MS        = 3000;  // cancel if no stillness within this window
const unsigned long FALL_COOLDOWN_MS      = 15000; // cooldown between fall alerts
const unsigned long PRIME_TIMEOUT_MS      = 3000;  // if nothing happens within 3s, drop primed state
const unsigned long REIMPACT_WINDOW_MS    = 1200;  // second strong impact allowed within 1.2s

// ================== STATE VARIABLES ==================
bool fallPrimed = false;            // we detected an impact, watching for fall
unsigned long firstImpactMs = 0;    // time of first impact
unsigned long lastImpactMs  = 0;    // time of last (possibly second) impact
unsigned long stillStartMs  = 0;    // when stillness started
unsigned long lastFallMs    = 0;    // last confirmed fall time

// Orientation (for debug only)
float rollDeg  = 0.0;
float pitchDeg = 0.0;
unsigned long lastMicros = 0;

// ================== ORIENTATION FILTER ==================
void updateOrientation(float ax, float ay, float az, float gx, float gy) {
  unsigned long nowMicros = micros();
  if (lastMicros == 0) lastMicros = nowMicros;
  float dt = (nowMicros - lastMicros) / 1e6f;
  lastMicros = nowMicros;

  float accelRoll  = atan2(ay, az) * 57.2958f;
  float accelPitch = atan2(-ax, sqrt(ay * ay + az * az)) * 57.2958f;

  static float rollFilt  = 0.0f;
  static float pitchFilt = 0.0f;

  rollFilt  = 0.98f * (rollFilt  + gx * 57.2958f * dt) + 0.02f * accelRoll;
  pitchFilt = 0.98f * (pitchFilt + gy * 57.2958f * dt) + 0.02f * accelPitch;

  rollDeg  = rollFilt;
  pitchDeg = pitchFilt;
}

// ================== SETUP ==================
void setup() {
  Serial.begin(115200);
  Wire.begin();

  if (!mpu.begin()) {
    Serial.println("MPU6050 not detected. Halting.");
    while (1) {
      delay(100);
    }
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  Serial.println("=== Fall Detection (Hand/Waist Wearable) ===");
}

// ================== MAIN LOOP ==================
void loop() {
  unsigned long nowMs = millis();

  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  // Acceleration magnitude in g
  float ax_g = a.acceleration.x / 9.81f;
  float ay_g = a.acceleration.y / 9.81f;
  float az_g = a.acceleration.z / 9.81f;
  float accelMag_g = sqrt(ax_g * ax_g + ay_g * ay_g + az_g * az_g);

  // Gyro magnitude in deg/s
  float gx_dps = g.gyro.x * 57.2958f;
  float gy_dps = g.gyro.y * 57.2958f;
  float gz_dps = g.gyro.z * 57.2958f;
  float gyroMag_dps = sqrt(gx_dps * gx_dps + gy_dps * gy_dps + gz_dps * gz_dps);

  // Update orientation for debugging
  updateOrientation(a.acceleration.x, a.acceleration.y, a.acceleration.z,
                    g.gyro.x, g.gyro.y);

  // --------- COOLDOWN: after a confirmed fall, wait before next ---------
  if (!fallPrimed && (nowMs - lastFallMs < FALL_COOLDOWN_MS)) {
    delay(50);
    return;
  }

  // --------- RESET PRIMED STATE IF TOO OLD ---------
  if (fallPrimed && (nowMs - firstImpactMs > PRIME_TIMEOUT_MS)) {
    Serial.println("Primed state timeout → no fall.");
    fallPrimed = false;
    stillStartMs = 0;
  }

  // ================== STEP 1: IMPACT DETECTION ==================
  bool normalImpact  = accelMag_g > IMPACT1_G;
  bool strongImpact  = accelMag_g > IMPACT2_G;
  bool rotationSpike = gyroMag_dps > GYRO_SPIKE_DPS;

  // First impact to prime a possible fall
  if (!fallPrimed) {
    if (strongImpact || (normalImpact && rotationSpike)) {
      fallPrimed     = true;
      firstImpactMs  = nowMs;
      lastImpactMs   = nowMs;
      stillStartMs   = 0;

      Serial.println("Impact detected → monitoring for fall...");
      Serial.print("  accelMag_g="); Serial.println(accelMag_g, 2);
      Serial.print("  gyroMag_dps="); Serial.println(gyroMag_dps, 2);
    }
  } else {
    // Already primed: allow a second stronger impact shortly after
    if (strongImpact && (nowMs - lastImpactMs < REIMPACT_WINDOW_MS)) {
      lastImpactMs = nowMs;
      Serial.println("Second strong impact detected while primed.");
    }
  }

  // If still not primed, nothing else to do this loop
  if (!fallPrimed) {
    delay(50);
    return;
  }

  // ================== STEP 2: CHECK FOR STILLNESS AFTER IMPACT ==================

  // Wait grace period so vibrations settle
  if (nowMs - lastImpactMs < GRACE_AFTER_IMPACT_MS) {
    delay(20);
    return;
  }

  // Check if sensor is "still" (after impact or re-impact)
  bool accelStill = (accelMag_g > STILL_MIN_G && accelMag_g < STILL_MAX_G);
  bool gyroStill  = (gyroMag_dps < STILL_GYRO_MAX);

  if (accelStill && gyroStill) {
    if (stillStartMs == 0) {
      stillStartMs = nowMs;
    }

    if (nowMs - stillStartMs >= STILL_CONFIRM_MS) {
      // ================== FALL CONFIRMED ==================
      Serial.println("✅ FALL CONFIRMED");
      Serial.print("  accelMag_g="); Serial.println(accelMag_g, 2);
      Serial.print("  gyroMag_dps="); Serial.println(gyroMag_dps, 2);
      Serial.print("  rollDeg="); Serial.println(rollDeg, 1);
      Serial.print("  pitchDeg="); Serial.println(pitchDeg, 1);
      Serial.println("----------------------------------");

      // TODO: HERE YOU TRIGGER YOUR ALERTS (buzzer, GSM, WiFi, etc.)

      lastFallMs   = nowMs;
      fallPrimed   = false;
      stillStartMs = 0;

      delay(500);  // short pause so we don't spam prints
    }
  } else {
    // If movement continues, reset still timer
    stillStartMs = 0;
  }

  // If we never get stillness within the fall window → cancel
  if (nowMs - firstImpactMs > FALL_WINDOW_MS && stillStartMs == 0) {
    Serial.println("Impact but no stabilized posture → not a fall.");
    fallPrimed = false;
  }

  delay(50); // ~20Hz loop
}
